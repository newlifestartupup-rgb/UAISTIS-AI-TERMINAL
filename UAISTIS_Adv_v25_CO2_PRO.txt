//======================================================================
// VWAP CALCULATIONS ENGINE
//======================================================================
vwapAnchorTime = vwapAnchor == "Session" ? timeframe.change("D") : vwapAnchor == "Week" ? timeframe.change("W") : vwapAnchor == "Month" ? timeframe.change("M") : timeframe.change("12M")

var float vwapSumSrcVol = 0.0
var float vwapSumVol    = 0.0
var float vwapSumSrc2Vol= 0.0

if vwapAnchorTime
    vwapSumSrcVol  := hlc3 * volume
    vwapSumVol     := volume
    vwapSumSrc2Vol := hlc3 * hlc3 * volume
else
    vwapSumSrcVol  += hlc3 * volume
    vwapSumVol     += volume
    vwapSumSrc2Vol += hlc3 * hlc3 * volume

vwapVal = vwapSumVol > 0 ? vwapSumSrcVol / vwapSumVol : hlc3
vwapVariance = math.max(0, (vwapSumSrc2Vol / vwapSumVol) - (vwapVal * vwapVal))
vwapStDev    = math.sqrt(vwapVariance)

vwapUpper1 = vwapVal + vwapStDev * 1.0
vwapLower1 = vwapVal - vwapStDev * 1.0
vwapUpper2 = vwapVal + vwapStDev * 2.0
vwapLower2 = vwapVal - vwapStDev * 2.0
vwapUpper3 = vwapVal + vwapStDev * 3.0
vwapLower3 = vwapVal - vwapStDev * 3.0

plot(showVWAP ? vwapVal : na, title="VWAP", color=c_vwap, linewidth=2)
pU1 = plot(showVWAP and showVWAPBands ? vwapUpper1 : na, title="VWAP Upper 1SD", color=c_band1, linewidth=1)
pL1 = plot(showVWAP and showVWAPBands ? vwapLower1 : na, title="VWAP Lower 1SD", color=c_band1, linewidth=1)

pL2 = plot(showVWAP and showVWAPBands ? vwapLower2 : na, title="VWAP Lower 2SD", color=c_band2, linewidth=1)

//======================================================================
// FOOTPRINT & ORDER FLOW CALCULATION ENGINE
//======================================================================
calcVolumeRatio = (high - low) > 0 ? (close - low) / (high - low) : 0.5
estimatedBuyVol  = volume * calcVolumeRatio
estimatedSellVol = volume * (1.0 - calcVolumeRatio)
barDeltaVolume   = estimatedBuyVol - estimatedSellVol

var float cumDeltaVolume = 0.0
if timeframe.change("D")
    cumDeltaVolume := barDeltaVolume
else
    cumDeltaVolume += barDeltaVolume

isBuyImbalance  = estimatedBuyVol >= (estimatedSellVol * imbalanceRatio)
isSellImbalance = estimatedSellVol >= (estimatedBuyVol * imbalanceRatio)

plotshape(showImbalance and isBuyImbalance, title="Buy Imbalance Footprint", style=shape.square, location=location.belowbar, color=c_buyDelta, size=size.tiny)
plotshape(showImbalance and isSellImbalance, title="Sell Imbalance Footprint", style=shape.square, location=location.abovebar, color=c_sellDelta, size=size.tiny)

//======================================================================
// CO2 THEORY (CANDLE 1/2) ENGINE
//======================================================================
co2PrevBody   = math.abs(close[1] - open[1])
co2PrevRange  = math.max(high[1] - low[1], syminfo.mintick)
co2PrevRatio  = co2PrevBody / co2PrevRange

co2CurrBody   = math.abs(close - open)
co2CurrRange  = math.max(high - low, syminfo.mintick)
co2CurrRatio  = co2CurrBody / co2CurrRange

bool co2Bullish = (close[1] > open[1]) and (close > open) and (co2CurrBody >= co2PrevBody * 0.8) and (co2CurrRatio >= co2MinBody)
bool co2Bearish = (close[1] < open[1]) and (close < open) and (co2CurrBody >= co2PrevBody * 0.8) and (co2CurrRatio >= co2MinBody)

plotshape(showCO2 and co2Bullish, title="CO2 Bullish Signal", style=shape.triangleup, location=location.belowbar, color=color.lime, text="CO2 BUY", textcolor=color.white, size=size.small)
plotshape(showCO2 and co2Bearish, title="CO2 Bearish Signal", style=shape.triangledown, location=location.abovebar, color=color.fuchsia, text="CO2 SELL", textcolor=color.white, size=size.small)

//======================================================================

//======================================================================
[diPlus, diMinus, adxVal] = ta.dmi(14, 14)
atrRatio = ta.atr(14) / ta.sma(ta.atr(14), 50)

bool isSideways = false
